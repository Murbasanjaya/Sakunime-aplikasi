# Membuat file APK Sakunime otomatis di GitHub.
# Jalan sendiri tiap ada perubahan di folder android/, atau manual lewat
# tab Actions -> "Build APK Sakunime" -> Run workflow.
# Hasil APK: buka run yang sudah selesai (centang hijau) -> bagian "Artifacts" -> sakunime-apk.
name: Build APK Sakunime

on:
  push:
    paths:
      - 'android/**'
      - '.github/workflows/android.yml'
  workflow_dispatch:

jobs:
  build:
    runs-on: ubuntu-latest
    steps:
      - uses: actions/checkout@v4

      - uses: actions/setup-java@v4
        with:
          distribution: temurin
          java-version: '17'

      - uses: android-actions/setup-android@v3

      - name: Pasang kunci tanda tangan (kalau secret sudah diisi)
        env:
          KEYSTORE_BASE64: ${{ secrets.KEYSTORE_BASE64 }}
        run: |
          if [ -n "$KEYSTORE_BASE64" ]; then
            mkdir -p android/keystore
            echo "$KEYSTORE_BASE64" | base64 -d > android/keystore/release.jks
            echo "Kunci rilis dipakai."
          else
            echo "::warning::Secret KEYSTORE_BASE64 belum diisi - APK ditandatangani kunci debug (update harus hapus aplikasi dulu)."
          fi

      - name: Build APK
        working-directory: android
        env:
          KEYSTORE_PASSWORD: ${{ secrets.KEYSTORE_PASSWORD }}
          KEY_ALIAS: ${{ secrets.KEY_ALIAS }}
          KEY_PASSWORD: ${{ secrets.KEY_PASSWORD }}
        run: |
          chmod +x gradlew
          ./gradlew --no-daemon assembleRelease

      - name: Ganti nama file APK
        run: |
          mkdir -p out
          cp android/app/build/outputs/apk/release/*.apk out/Sakunime.apk

      - uses: actions/upload-artifact@v4
        with:
          name: sakunime-apk
          path: out/Sakunime.apk
          retention-days: 30
